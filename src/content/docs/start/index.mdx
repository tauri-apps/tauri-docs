function showTab(id) {
  document.querySelectorAll('.tab').forEach(tab => tab.style.display='none');
  document.getElementById(id).style.display='block';
  if(id==="center") renderCentre();
}

// Sauvegarde générique
function saveData(key, data) {
  const arr = JSON.parse(localStorage.getItem(key)||'[]');
  arr.push(data);
  localStorage.setItem(key, JSON.stringify(arr));
}

// Affichage générique
function renderList(key, listId, formatter) {
  const arr = JSON.parse(localStorage.getItem(key)||'[]');
  document.getElementById(listId).innerHTML =
    arr.map((item,idx) => `<li>${formatter(item,idx)}</li>`).join('');
}

// Auditions
function saveAudition(){
  const txt=audText.value;

}

// Urgences
function saveUrgence(){
  saveData('urg',{num:urgNumber.value,loc:urgLoc.value,msg:urgMsg.value,date:new Date().toLocaleString()});
  urgNumber.value=''; urgLoc.value=''; urgMsg.value='';

}

// Centre Alertes
function renderCentre(){
  renderList('urg','centerList',(i,idx)=>`
    ${i.date}: ${i.num} at ${i.loc} - ${i.msg}
    <select onchange="dispatch(${idx},this.value)">
      <option value="">Affecter</option>
      <option>Police</option><option>Pompiers</option><option>SAMU</option>
    </select>
    <span id="stat${idx}">${i.dispatched||''}</span>
  `);
}
function dispatch(idx, unit){
  const arr = JSON.parse(localStorage.getItem('urg')||'[]');
  arr[idx].dispatched = unit;
  localStorage.setItem('urg', JSON.stringify(arr));
  document.getElementById(`stat${idx}`).innerText = unit;
}
renderCentre();
